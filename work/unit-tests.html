// the chime plays on a rise, never on first load
it("plays only when the unread count goes up", () => {
  const { rerender } = renderHook(
    ({ count }) => useNotificationAudio({ unreadCount: count }),
    { initialProps: { count: 3 }, wrapper },
  );
  expect(play).not.toHaveBeenCalled();

  rerender({ count: 5 });
  expect(play).toHaveBeenCalledTimes(1);

  rerender({ count: 4 });
  expect(play).toHaveBeenCalledTimes(1);
});

it("ignores a blocked autoplay", async () => {
  play.mockRejectedValueOnce(
    new DOMException("blocked", "NotAllowedError"));
  const { rerender } = renderHook(...);
  rerender({ count: 9 });
  await waitFor(() =>
    expect(console.error).not.toHaveBeenCalled());
});
